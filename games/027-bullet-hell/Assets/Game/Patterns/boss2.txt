# 5. ゆっくり出て加速する花びらの輪と、速い自機狙い。
pattern "Glass Petals" hp=800 time=40s
  move wander area=110,320,274,390 over=100
  emit ring every=36 count=24 angle=rand(0,15) speed=0.2 accel=0.03 maxspeed=2.8 shape=orb color=cyan
  emit aimed every=24 count=3 spread=16 speed=3.2 shape=small color=white
